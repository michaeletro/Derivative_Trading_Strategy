cmake_minimum_required(VERSION 3.16)
project(DerivativeTradingCore VERSION 0.2.0 LANGUAGES CXX)
option(DTS_BUILD_SERVER "Build the owned Crow/SQLite read-only service" OFF)
option(DTS_WITH_IBKR "Build the optional official-SDK native TWS adapter" OFF)
add_library(trading_core INTERFACE)
target_compile_features(trading_core INTERFACE cxx_std_17)
target_include_directories(trading_core INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/src/backend/cpp_src/core/include)
function(configure_executable target)
    target_link_libraries(${target} PRIVATE trading_core)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()
add_executable(broker_demo apps/broker_demo.cpp)
configure_executable(broker_demo)
include(CTest)
if(BUILD_TESTING)
    foreach(suite core_tests tws_state_tests)
        add_executable(${suite} tests/${suite}.cpp)
        configure_executable(${suite})
        add_test(NAME ${suite} COMMAND ${suite})
    endforeach()
endif()
if(DTS_WITH_IBKR)
    include(cmake/IBKR.cmake)
endif()
if(DTS_BUILD_SERVER)
    find_package(Threads REQUIRED)
    find_package(Boost REQUIRED COMPONENTS system)
    find_package(SQLite3 REQUIRED)
    add_executable(server src/backend/cpp_src/src/main.cpp src/backend/cpp_src/src/Utilities/server.cpp)
    configure_executable(server)
    target_link_libraries(server PRIVATE Boost::system SQLite::SQLite3 Threads::Threads)
    # Retained bundled Crow version; vendor warnings are not application errors.
    target_include_directories(server SYSTEM PRIVATE src/backend/cpp_src/headers/Utilities)
    if(DTS_WITH_IBKR)
        target_compile_definitions(server PRIVATE DTS_WITH_IBKR=1)
        target_link_libraries(server PRIVATE trading_ibkr)
    endif()
    if(BUILD_TESTING)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        add_test(NAME http_smoke_tests COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/http_smoke_tests.py" "$<TARGET_FILE:server>")
        set_tests_properties(http_smoke_tests PROPERTIES TIMEOUT 25)
    endif()
endif()
